# Release Index, Detail, and Book Rendering Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Implement Synctrol Release index and detail surfaces — including independent cover/artwork/album-cover image roles, Diverse System–style square artwork grids, optional Album/Gift Book bodies, MusicAlbum/MusicRecording JSON-LD for album books, and draft badges when `showDrafts` is enabled.

**Architecture:** Pure Node/shared modules assemble Release list and detail view models from already-compiled routes, content packages, and resolved assets. Vue layouts and components under `src/client/` render those models inside the existing shell. Platforms render through the existing platform system; assets resolve through the existing asset pipeline. This plan does not reimplement discovery, routes, shell chrome, or platform validation.

**Tech Stack:** TypeScript, Vitest, Vue 3, `@vue/test-utils`, `happy-dom`, VuePress 2 theme package `vuepress-theme-synctrolling` (Plans 01–07 assumed complete for types, compiler, routes, assets, shell, platforms).

## Global Constraints

- Package name: `vuepress-theme-synctrolling`
- Content types remain only `home | release | news | page`
- Brand tokens are fixed: black/white, `3px` strong border, `0` radius, `--syn-artwork-width: 660px`, `--syn-content-width: 760px`
- There is no Release artwork-loading option and no fallback among `cover`, `artwork`, and `book.yml → album.covers`
- `release.index.enabled: false` suppresses Release Index and its pagination, but not Release detail pages
- Release index defaults: `pagination: 12`, `mobileGridColumns: 2`, `desktopGridColumns: 3`; `desktopGridColumns` ∈ 1–6; `mobileGridColumns` ∈ 1–3
- Release index sorts by date descending (then slug for stability); page one uses the index route; numbered pages start at 2
- Missing artwork uses `release.artworkPlaceholder` when configured, otherwise a branded empty frame; never `cover`
- Album Book body order: platform links → album covers → discs/tracks; display numbering is 1-based; track `duration` is non-negative integer seconds
- Gift Book body: items in array order; each item renders its own covers then its own links; Gift links are never hoisted
- A Release without `book.yml` remains valid, appears in the index, and renders Markdown-only detail
- JSON-LD `MusicAlbum` / `MusicRecording` only for Album books; Gift emits no `Product`
- Draft badges appear on list tiles and detail when `showDrafts` makes drafts visible
- Tests run with `pnpm exec vitest run <path>`
- Plans 01–07 are assumed complete: shared types, Book validation, locale routes/collections, assets, shell, platforms

```ts
// src/shared/release/numbering.ts
import type { Disc, Multilanguage, Track } from '../types'

export interface NumberedTrack {
  number: number
  anchor: string
  title: Multilanguage
  artists: string[]
  durationSeconds: number
  durationLabel: string
  desc?: Multilanguage
  copyright?: string
}

export interface NumberedDisc {
  number: number
  anchor: string
  title: Multilanguage
  desc?: Multilanguage
  tracks: NumberedTrack[]
}

export function formatTrackDuration(seconds: number): string {
  const whole = Math.max(0, Math.floor(seconds))
  const minutes = Math.floor(whole / 60)
  const rem = whole % 60
  return `${minutes}:${String(rem).padStart(2, '0')}`
}

export function formatMessage(
  template: string,
  vars: Record<string, string | number>,
): string {
  return template.replace(/\{(\w+)\}/g, (_, key: string) => String(vars[key] ?? ''))
}

export function numberDiscs(discs: Disc[]): NumberedDisc[] {
  return discs.map((disc, discIndex) => {
    const discNumber = discIndex + 1
    const discAnchor = `disc-${discNumber}`
    return {
      number: discNumber,
      anchor: discAnchor,
      title: disc.title,
      ...(disc.desc !== undefined ? { desc: disc.desc } : {}),
      tracks: disc.tracks.map((track: Track, trackIndex) => {
        const trackNumber = trackIndex + 1
        return {
          number: trackNumber,
          anchor: `${discAnchor}-track-${trackNumber}`,
          title: track.title,
          artists: track.artists,
          durationSeconds: track.duration,
          durationLabel: formatTrackDuration(track.duration),
          ...(track.desc !== undefined ? { desc: track.desc } : {}),
          ...(track.copyright !== undefined
            ? { copyright: track.copyright }
            : {}),
        }
      }),
    }
  })
}
```

```ts
// src/node/release/list-model.ts
import type { ContentPackage, ReleaseOptions } from '../../shared/types'
import type { CompiledPage, ContentIdentity } from '../../shared/types/routes'
import type { ResolvedAsset } from '../assets/types'
import type {
  ReleaseArtworkKind,
  ReleaseIndexModel,
  ReleaseIndexTile,
} from '../../shared/release/types'
import { selectReleaseArtwork } from '../../shared/release/image-roles'

export interface BuildReleaseIndexModelInput {
  collectionPage: CompiledPage
  detailPages: CompiledPage[]
  packages: ContentPackage[]
  releaseOptions: ReleaseOptions
  resolveArtwork: (pkg: ContentPackage) => ResolvedAsset | undefined
  resolvePlaceholder: () => ResolvedAsset | undefined
  showDrafts: boolean
}

function packageByIdentity(
  packages: ContentPackage[],
  identity: ContentIdentity,
): ContentPackage | undefined {
  if (!identity.startsWith('release:')) return undefined
  const slug = identity.slice('release:'.length)
  return packages.find((p) => p.type === 'release' && p.slug === slug)
}

export function buildReleaseIndexModel(
  input: BuildReleaseIndexModelInput,
): ReleaseIndexModel | null {
  if (!input.releaseOptions.index.enabled) return null
  if (input.collectionPage.contentType !== 'release-collection') {
    throw new Error('buildReleaseIndexModel requires a release-collection page')
  }
  const collection = input.collectionPage.collection
  if (!collection) {
    throw new Error('release-collection page missing collection metadata')
  }

  const detailByIdentity = new Map(
    input.detailPages.map((p) => [p.identity, p] as const),
  )

  const tiles: ReleaseIndexTile[] = []
  for (const identity of collection.itemIdentities) {
    const detail = detailByIdentity.get(identity)
    const pkg = packageByIdentity(input.packages, identity)
    if (!detail || !pkg || pkg.type !== 'release') continue

    const artworkPath = selectReleaseArtwork({
      cover: pkg.cover,
      artwork: pkg.artwork,
    })
    const resolvedArtwork = artworkPath
      ? input.resolveArtwork(pkg)
      : undefined
    const placeholder = !resolvedArtwork
      ? input.releaseOptions.artworkPlaceholder
        ? input.resolvePlaceholder()
        : undefined
      : undefined

    let artworkKind: ReleaseArtworkKind
    let artwork: ResolvedAsset | undefined
    if (resolvedArtwork) {
      artworkKind = 'artwork'
      artwork = resolvedArtwork
    } else if (placeholder) {
      artworkKind = 'placeholder'
      artwork = placeholder
    } else {
      artworkKind = 'empty-frame'
      artwork = undefined
    }

    tiles.push({
      identity: identity as `release:${string}`,
      slug: pkg.slug!,
      title: detail.title,
      date: pkg.date!,
      href: detail.url.publicPath,
      artwork,
      artworkKind,
      isDraft: detail.isDraft,
      showDraftBadge: Boolean(input.showDrafts && detail.isDraft),
      isFallback: detail.isFallback,
      showDate: false,
      showDescription: false,
      accessibleName: detail.title,
    })
  }

  return {
    locale: input.collectionPage.locale,
    page: collection.page,
    pageCount: collection.pageCount,
    mobileGridColumns: input.releaseOptions.index.mobileGridColumns,
    desktopGridColumns: input.releaseOptions.index.desktopGridColumns,
    tiles,
    empty: tiles.length === 0,
  }
}
```

```ts
// src/node/release/jsonld-album.ts
import type { Book, LocaleKey } from '../../shared/types'
import { resolveMultilanguage } from '../../shared/multilanguage'
import { numberDiscs } from '../../shared/release/numbering'

export interface BuildAlbumJsonLdInput {
  book: Book | undefined
  locale: LocaleKey
  mainLocale: LocaleKey
  pageAbsoluteUrl: string
  datePublished: string
}

function iso8601Duration(seconds: number): string {
  const whole = Math.max(0, Math.floor(seconds))
  const minutes = Math.floor(whole / 60)
  const rem = whole % 60
  if (minutes === 0) return `PT${rem}S`
  if (rem === 0) return `PT${minutes}M`
  return `PT${minutes}M${rem}S`
}

export function buildAlbumJsonLd(input: BuildAlbumJsonLdInput): object[] {
  const { book } = input
  if (!book || book.type !== 'album') return []

  const name = resolveMultilanguage(book.title, input.locale, input.mainLocale).text
  const numbered = numberDiscs(book.album.discs ?? [])
  const track = numbered.flatMap((disc) =>
    disc.tracks.map((t) => ({
      '@type': 'MusicRecording',
      name: resolveMultilanguage(t.title, input.locale, input.mainLocale).text,
      position: t.number,
      duration: iso8601Duration(t.durationSeconds),
      byArtist: t.artists.map((artist) => ({
        '@type': 'MusicGroup',
        name: artist,
      })),
    })),
  )

  const album: Record<string, unknown> = {
    '@context': 'https://schema.org',
    '@type': 'MusicAlbum',
    name,
    url: input.pageAbsoluteUrl,
    datePublished: input.datePublished,
    track,
  }

  if (book.authors?.length) {
    album.byArtist = book.authors.map((artist) => ({
      '@type': 'MusicGroup',
      name: artist,
    }))
  }

  if (book.desc) {
    album.description = resolveMultilanguage(
      book.desc,
      input.locale,
      input.mainLocale,
    ).text
  }

  return [album]
}
```

```ts
// src/node/release/detail-model.ts
import type {
  Book,
  ContentPackage,
  LocaleKey,
  LocaleMessages,
  ReleaseOptions,
} from '../../shared/types'
import type { CompiledPage } from '../../shared/types/routes'
import type { ResolvedAsset } from '../assets/types'
import type { ReleaseDetailModel, ReleaseDetailSection } from '../../shared/release/types'
import { selectAlbumCovers, selectReleaseArtwork } from '../../shared/release/image-roles'
import { numberDiscs } from '../../shared/release/numbering'
import { resolveMultilanguage } from '../../shared/multilanguage'
import { buildAlbumJsonLd } from './jsonld-album'

export interface BuildReleaseDetailModelInput {
  page: CompiledPage
  pkg: ContentPackage
  book: Book | undefined
  messages: LocaleMessages
  mainLocale: LocaleKey
  releaseIndexHref: string
  markdownHtml: string
  resolveArtwork: (pkg: ContentPackage) => ResolvedAsset | undefined
  resolveAlbumCover: (relativePath: string) => ResolvedAsset
  resolveGiftItemCover?: (relativePath: string) => ResolvedAsset
  resolvePlaceholder: () => ResolvedAsset | undefined
  releaseOptions: ReleaseOptions
  showDrafts: boolean
  formatDate: (yyyyMmDd: string, locale: LocaleKey) => string
}

function resolvedText(
  value: import('../../shared/types').Multilanguage,
  locale: LocaleKey,
  mainLocale: LocaleKey,
) {
  const r = resolveMultilanguage(value, locale, mainLocale)
  return {
    text: r.text,
    ...(r.fellBack ? { lang: r.locale } : {}),
  }
}

export function buildReleaseDetailModel(
  input: BuildReleaseDetailModelInput,
): ReleaseDetailModel {
  const { page, pkg, book, messages } = input
  const sections: ReleaseDetailSection[] = []

  sections.push({
    kind: 'return-link',
    href: input.releaseIndexHref,
    label: messages.returnToReleases,
  })

  sections.push({
    kind: 'title-date',
    title: page.title,
    date: input.formatDate(pkg.date!, page.locale),
    dateLabel: messages.published,
    ...(page.isFallback ? { titleLang: page.bodyLocale } : {}),
  })

  const artworkPath = selectReleaseArtwork({
    cover: pkg.cover,
    artwork: pkg.artwork,
  })
  const resolvedArtwork = artworkPath ? input.resolveArtwork(pkg) : undefined
  const placeholder =
    !resolvedArtwork && input.releaseOptions.artworkPlaceholder
      ? input.resolvePlaceholder()
      : undefined
  sections.push({
    kind: 'artwork',
    artworkKind: resolvedArtwork
      ? 'artwork'
      : placeholder
        ? 'placeholder'
        : 'empty-frame',
    artwork: resolvedArtwork ?? placeholder,
    alt: page.title,
  })

  if (book) {
    sections.push({
      kind: 'book-identity',
      bookType: book.type,
      title: resolvedText(book.title, page.locale, input.mainLocale),
      ...(book.desc
        ? { desc: resolvedText(book.desc, page.locale, input.mainLocale) }
        : {}),
      ...(book.authors ? { authors: book.authors } : {}),
      ...(book.copyright ? { copyright: book.copyright } : {}),
    })

    if (book.type === 'album') {
      const covers = selectAlbumCovers({ artwork: pkg.artwork, book }).map((p) =>
        input.resolveAlbumCover(p),
      )
      sections.push({
        kind: 'album-body',
        order: ['links', 'covers', 'discs'],
        links: book.album.links ?? [],
        covers,
        discs: numberDiscs(book.album.discs ?? []),
        labels: {
          platformLinks: messages.platformLinks,
          covers: messages.covers,
          tracklist: messages.tracklist,
          disc: messages.disc,
          track: messages.track,
        },
      })
    } else {
      const resolveCover = input.resolveGiftItemCover ?? input.resolveAlbumCover
      sections.push({
        kind: 'gift-body',
        items: book.gift.items.map((item) => ({
          id: item.id,
          title: resolvedText(item.title, page.locale, input.mainLocale),
          ...(item.desc
            ? { desc: resolvedText(item.desc, page.locale, input.mainLocale) }
            : {}),
          covers: (item.covers ?? []).map((p) => resolveCover(p)),
          links: item.links ?? [],
          ...(item.copyright ? { copyright: item.copyright } : {}),
          coverOrder: 'before-links' as const,
          linksHoisted: false as const,
        })),
        labels: {
          giftItems: messages.giftItems,
          covers: messages.covers,
          platformLinks: messages.platformLinks,
        },
      })
    }
  }

  sections.push({
    kind: 'markdown',
    html: input.markdownHtml,
    bodyLang: page.bodyLocale,
  })

  return {
    sections,
    jsonLd: buildAlbumJsonLd({
      book,
      locale: page.locale,
      mainLocale: input.mainLocale,
      pageAbsoluteUrl: page.url.absoluteUrl,
      datePublished: pkg.date!,
    }),
    showDraftBadge: Boolean(input.showDrafts && page.isDraft),
    draftLabel: messages.draft,
    includedInIndex: true,
  }
}
```
